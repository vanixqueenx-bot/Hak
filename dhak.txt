-- وظيفة مسح المحتوى الحالي لإظهار القسم المطلوب فقط
local function clearContent()
    for _, child in ipairs(ContentContainer:GetChildren()) do
        if child:IsA("GuiObject") then
            child:Destroy()
        end
    end
end

-- وظيفة لإنشاء خيار أو خاصية داخل القسم
local function addOption(titleText, defaultState, callback)
    local yPos = #ContentContainer:GetChildren() * 45
    
    local optionFrame = Instance.new("Frame")
    optionFrame.Parent = ContentContainer
    optionFrame.BackgroundColor3 = Color3.fromRGB(38, 38, 48)
    optionFrame.Position = UDim2.new(0, 5, 0, yPos)
    optionFrame.Size = UDim2.new(0, 325, 0, 40)
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 4)
    corner.Parent = optionFrame
    
    local label = Instance.new("TextLabel")
    label.Parent = optionFrame
    label.BackgroundTransparency = 1
    label.Position = UDim2.new(0, 10, 0, 0)
    label.Size = UDim2.new(0, 240, 1, 0)
    label.Font = Enum.Font.SourceSansBold
    label.Text = titleText
    label.TextColor3 = Color3.fromRGB(240, 240, 240)
    label.TextSize = 13
    label.TextXAlignment = Enum.TextXAlignment.Left
    
    local toggleBtn = Instance.new("TextButton")
    toggleBtn.Parent = optionFrame
    toggleBtn.BackgroundColor3 = defaultState and Color3.fromRGB(40, 180, 80) or Color3.fromRGB(180, 40, 40)
    toggleBtn.Position = UDim2.new(1, -55, 0, 8)
    toggleBtn.Size = UDim2.new(0, 45, 0, 24)
    toggleBtn.Font = Enum.Font.SourceSansBold
    toggleBtn.Text = defaultState and "ON" or "OFF"
    toggleBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    toggleBtn.TextSize = 11
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 4)
    btnCorner.Parent = toggleBtn
    
    local state = defaultState
    toggleBtn.MouseButton1Click:Connect(function()
        state = not state
        toggleBtn.BackgroundColor3 = state and Color3.fromRGB(40, 180, 80) or Color3.fromRGB(180, 40, 40)
        toggleBtn.Text = state and "ON" or "OFF"
        if callback then callback(state) end
    end)
end

-- المتغيرات للتحكم بالخصائص
local autoStealEnabled = false
local antiTrapEnabled = false

-- 1. قسم الـ Farm (السرقة والحماية التلقائية)
addSidebarButton("FarmBtn", "Farm ⚡", function()
    clearContent()
    
    addOption("Auto Steal Eggs (سرقة البيض تلقائياً)", autoStealEnabled, function(val)
        autoStealEnabled = val
        task.spawn(function()
            while autoStealEnabled do
                task.wait(0.5)
                pcall(function()
                    local char = LocalPlayer.Character
                    if char and char:FindFirstChild("HumanoidRootPart") then
                        for _, obj in ipairs(Workspace:GetChildren()) do
                            if not autoStealEnabled then break end
                            if string.find(obj.Name:lower(), "egg") then
                                local part = obj:FindFirstChildWhichIsA("BasePart")
                                if part then
                                    char.HumanoidRootPart.CFrame = part.CFrame + Vector3.new(0, 2, 0)
                                end
                            end
                        end
                    end
                end)
            end
        end)
    end)
    
    addOption("Anti-Trap & Bat (حماية من الفخاخ والمضرب)", antiTrapEnabled, function(val)
        antiTrapEnabled = val
        task.spawn(function()
            while antiTrapEnabled do
                task.wait(0.3)
                pcall(function()
                    local char = LocalPlayer.Character
                    if char and char:FindFirstChild("HumanoidRootPart") then
                        for _, trap in ipairs(Workspace:GetChildren()) do
                            if string.find(trap.Name:lower(), "trap") or string.find(trap.Name:lower(), "bat") then
                                local part = trap:FindFirstChildWhichIsA("BasePart")
                                if part and (part.Position - char.HumanoidRootPart.Position).Magnitude < 12 then
                                    char.HumanoidRootPart.CFrame = char.HumanoidRootPart.CFrame + Vector3.new(6, 0, 6)
                                end
                            end
                        end
                    end
                end)
            end
        end)
    end)
end)
