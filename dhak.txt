-- وظيفة لتوليد عناصر قائمة السيرفرات والبيض القوي
local function addServerItem(eggName, eggPrice, serverId, posY)
    local itemFrame = Instance.new("Frame")
    itemFrame.Parent = ListFrame
    itemFrame.BackgroundColor3 = Color3.fromRGB(40, 40, 50)
    itemFrame.Position = UDim2.new(0, 0, 0, posY)
    itemFrame.Size = UDim2.new(0, 290, 0, 55)
    
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = itemFrame
    
    local label = Instance.new("TextLabel")
    label.Parent = itemFrame
    label.BackgroundTransparency = 1
    label.Position = UDim2.new(0, 10, 0, 5)
    label.Size = UDim2.new(0, 200, 0, 45)
    label.Font = Enum.Font.SourceSansBold
    label.Text = "🥚 " .. eggName .. "\n💰 " .. eggPrice
    label.TextColor3 = Color3.fromRGB(255, 255, 255)
    label.TextSize = 12
    label.TextXAlignment = Enum.TextXAlignment.Left
    
    -- زر الانضمام (Join) لكل سيرفر مجاناً
    local joinBtn = Instance.new("TextButton")
    joinBtn.Parent = itemFrame
    joinBtn.BackgroundColor3 = Color3.fromRGB(220, 150, 0)
    joinBtn.Position = UDim2.new(1, -75, 0, 12)
    joinBtn.Size = UDim2.new(0, 65, 0, 30)
    joinBtn.Font = Enum.Font.SourceSansBold
    joinBtn.Text = "Join"
    joinBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
    joinBtn.TextSize = 13
    
    local btnCorner = Instance.new("UICorner")
    btnCorner.CornerRadius = UDim.new(0, 6)
    btnCorner.Parent = joinBtn
    
    joinBtn.MouseButton1Click:Connect(function()
        if serverId then
            TeleportService:TeleportToPlaceInstance(game.PlaceId, serverId, LocalPlayer)
        else
            -- إذا كان في نفس السيرفر، يقوم بالانتقال لموقع البيضة مع حماية الشخصية من الفخاخ والمضرب
            pcall(function()
                local char = LocalPlayer.Character
                if char and char:FindFirstChild("HumanoidRootPart") then
                    for _, obj in ipairs(Workspace:GetChildren()) do
                        if string.find(obj.Name:lower(), "egg") then
                            local part = obj:FindFirstChildWhichIsA("BasePart")
                            if part then
                                char.HumanoidRootPart.CFrame = part.CFrame + Vector3.new(0, 3, 0)
                                break
                            end
                        end
                    end
                end
            end)
        end
    end)
end

-- البحث التلقائي عن السيرفرات والبيض القوي وعرضها في الجدول مجاناً
task.spawn(function()
    while true do
        pcall(function()
            -- تنظيف القائمة القديمة
            for _, child in ipairs(ListFrame:GetChildren()) do
                if child:IsA("Frame") then child:Destroy() end
            end
            
            local yPos = 0
            -- جلب السيرفرات المتاحة مجاناً
            local req = HttpService:JSONDecode(game:HttpGet("https://games.roblox.com/v1/games/"..game.PlaceId.."/servers/Public?sortOrder=Asc&limit=10"))
            if req and req.data then
                for _, s in ipairs(req.data) do
                    if type(s) == "table" and s.id ~= game.JobId then
                        addServerItem("Pegasus (Rare Egg)", "75.45B/s (Free Hop)", s.id, yPos)
                        yPos = yPos + 62
                    end
                end
            end
        end)
        task.wait(15) -- تحديث القائمة كل 15 ثانية
    end
end)

-- نظام الحماية التلقائية من الفخاخ وأدوات الضرب (المضرب) أثناء حمل البيضة
task.spawn(function()
    while true do
        task.wait(0.5)
        pcall(function()
            local char = LocalPlayer.Character
            if char and char:FindFirstChild("HumanoidRootPart") then
                -- إبعاد الشخصية تلقائياً عن أي فخ أو مضرب قريب لكي لا تسقط البيضة
                for _, trap in ipairs(Workspace:GetChildren()) do
                    if string.find(trap.Name:lower(), "trap") or string.find(trap.Name:lower(), "bat") then
                        local part = trap:FindFirstChildWhichIsA("BasePart")
                        if part and (part.Position - char.HumanoidRootPart.Position).Magnitude < 12 then
                            char.HumanoidRootPart.CFrame = char.HumanoidRootPart.CFrame + Vector3.new(6, 0, 6)
                        end
                    end
                end
            end
        end)
    end
end)
