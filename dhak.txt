-- وظيفة آمنة للبحث عن البيض بدون تجميد اللعبة (باستخدام pcall وتجنب GetDescendants الثقيلة)
local finderActive = false
FinderBtn.MouseButton1Click:Connect(function()
    finderActive = not finderActive
    if finderActive then
        FinderBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 100)
        StatusLabel.Text = "الحالة: جاري البحث... 🔍"
        
        task.spawn(function()
            while finderActive do
                task.wait(2) -- زيادة المهلة لتخفيف الضغط على المعالج
                
                -- حماية الحلقة بـ pcall لمنع انهيار السكربت عند حذف كائن مفاجئ
                local success, err = pcall(function()
                    for _, obj in ipairs(Workspace:GetChildren()) do -- استخدام GetChildren على المجلدات الرئيسية بدلاً من GetDescendants الشاملة لتوفير الأداء
                        if not finderActive then break end
                        
                        if obj:IsA("Model") and string.find(obj.Name:lower(), "egg") then
                            local part = obj.PrimaryPart or obj:FindFirstChildWhichIsA("BasePart")
                            local char = LocalPlayer.Character
                            
                            if part and char and char:FindFirstChild("HumanoidRootPart") then
                                StatusLabel.Text = "تم العثور على بيضة! جاري الانتقال..."
                                char.HumanoidRootPart.CFrame = part.CFrame + Vector3.new(0, 3, 0)
                                task.wait(3)
                                StatusLabel.Text = "الحالة: جاري البحث... 🔍"
                                break
                            end
                        end
                    end
                end)
                
                if not success then
                    warn("خطأ في حلقة البحث: ", err)
                end
            end
        end)
    else
        FinderBtn.BackgroundColor3 = Color3.fromRGB(50, 150, 250)
        StatusLabel.Text = "الحالة: متوقف 🛑"
    end
end)

-- منطق السرقة التلقائية مع حماية pcall
local autoStealActive = false
AutoStealBtn.MouseButton1Click:Connect(function()
    autoStealActive = not autoStealActive
    if autoStealActive then
        AutoStealBtn.BackgroundColor3 = Color3.fromRGB(50, 200, 100)
        AutoStealStatus.Text = "السرقة التلقائية: تعمل 🟢"
        
        task.spawn(function()
            while autoStealActive do
                task.wait(1)
                pcall(function()
                    for _, obj in ipairs(Workspace:GetChildren()) do
                        if not autoStealActive then break end
                        if string.find(obj.Name:lower(), "egg") then
                            local touchPart = obj:FindFirstChildWhichIsA("BasePart")
                            local rootPart = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
                            
                            if touchPart and rootPart and firetouchinterest then
                                firetouchinterest(rootPart, touchPart, 0)
                                firetouchinterest(rootPart, touchPart, 1)
                            end
                        end
                    end
                end)
            end
        end)
    else
        AutoStealBtn.BackgroundColor3 = Color3.fromRGB(50, 150, 250)
        AutoStealStatus.Text = "السرقة التلقائية: معطلة ❌"
    end
end)
